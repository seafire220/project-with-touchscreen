# --- Character device (added by device/install.sh) ---
# No rainbow screen at power-on.
disable_splash=1
# The listen button (optional on touchscreen builds): GPIO17 (pin 11) to ground (pin 9). Sends key 148 (KEY_PROG1).
dtoverlay=gpio-key,gpio=17,active_low=1,gpio_pull=up,keycode=148,label=listen
# A microphone mute switch, if fitted: a slide switch from GPIO27 (pin 13) to ground (pin 14 or 20),
# closed for "off". Sends key 248 (KEY_MICMUTE); remove the # to use it.
#dtoverlay=gpio-key,gpio=27,active_low=1,gpio_pull=up,keycode=248,label=mute
# The USB-C port as a USB device, so a computer can connect by cable (device/usb/companion-usb-gadget).
dtoverlay=dwc2,dr_mode=peripheral
# The power button: GPIO3 (pin 5) to ground (pin 14). Press to shut down; press again to start.
dtoverlay=gpio-shutdown,gpio_pin=3,active_low=1,gpio_pull=up
# Raspberry Pi Touch Display 2, 5 inch, on the DSI connector (display + capacitive touch).
# For the 7 inch version use vc4-kms-dsi-ili9881-7inch. Check the Raspberry Pi documentation
# for your display; an HDMI screen needs neither line (USB touchscreens still work via libinput).
dtoverlay=vc4-kms-dsi-ili9881-5inch
# --- end of character device settings ---
